// Factory with closures
function createUser(role) {
  const permissions = role === 'admin'
    ? ['read', 'write', 'delete']
    : ['read'];
  return { role, permissions };
}

// Proxy — lazy loading + caching
const handler = {
  get(target, prop) {
    if (!(prop in target)) {
      target[prop] = expensiveComputation(prop);
    }
    return target[prop];
  }
};
const cache = new Proxy({}, handler);

// Decorator — higher-order function
function withLogging(fn) {
  return function(...args) {
    console.log(`Calling ${fn.name}(${args})`);
    const result = fn.apply(this, args);
    console.log(`Result: ${result}`);
    return result;
  };
}

bus.on('user:created', (user) => sendWelcomeEmail(user));

// Strategy pattern
const strategies = {
  bubble: (arr) => { /* bubble sort */ },
  quick: (arr) => { /* quick sort */ },
  merge: (arr) => { /* merge sort */ },
};
function sort(arr, algorithm = 'quick') {
  return strategies[algorithm](arr);
}

GRASP Principles

  • Information Expert — assign responsibility to the class with the most information
  • Creator — assign creation to the class that has the data needed
  • Controller — delegate system events to a non-UI object
  • Low Coupling — minimize dependencies between modules
  • High Cohesion — keep related behavior together
  • Polymorphism — use interfaces/protocols instead of conditionals
  • Pure Fabrication — create utility classes when no domain object fits
  • Indirection — add intermediate objects to reduce coupling
  • Protected Variations — hide instability behind stable interfaces

Anti-Patterns

  • God Object — one module that does everything
  • Callback Hell — deeply nested callbacks (use async/await)
  • Premature Optimization — optimizing before measuring
  • Copy-Paste Programming — duplication instead of abstraction
  • Magic Numbers — unexplained numeric literals
  • Global State — shared mutable state across modules
